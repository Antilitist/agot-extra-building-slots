# Holding auto-upgrade. No engine construct-upgrade effect in 1.19.
# Charge gold, wait real construction months (~standard_construction_time / 30), then add_building.
# Optional GUI capture stores jen_auto_cost_factor / jen_auto_time_factor on the province.

jen_auto_upgrade_pay_and_queue_effect = {
	# Scale script cost/time if this holding captured GUITrackItem next cost/time.
	set_variable = {
		name = jen_auto_this_cost
		value = {
			value = $COST$
			if = {
				limit = { has_variable = jen_auto_cost_factor }
				multiply = var:jen_auto_cost_factor
			}
			min = 1
		}
	}
	if = {
		limit = {
			NOT = { has_variable = jen_did_upgrade }
			NOT = { has_variable = jen_auto_queue_building }
			has_ongoing_construction = no
			exists = province_owner
			province_owner = { gold >= var:jen_auto_this_cost }
		}
		province_owner = {
			remove_short_term_gold = var:jen_auto_this_cost
		}
		set_variable = { name = jen_auto_queue_building value = flag:$BUILDING$ }
		set_variable = { name = jen_auto_queue_months value = jen_auto_upgrade_wait_months }
		set_variable = { name = jen_did_upgrade value = yes }
		debug_log = "BotD auto-upgrade queued $BUILDING$"
		province_owner = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				# $BUILDING$ expands per call (e.g. building_outposts_02)
				title = building_$BUILDING$
				left_icon = this
				custom_tooltip = jen_auto_upgrade_started_toast_tt
			}
		}
	}
	remove_variable = jen_auto_this_cost
}

	if = {
		limit = { has_variable = jen_auto_queue_building }
		if = {
			limit = {
				has_variable = jen_auto_queue_months
				var:jen_auto_queue_months > 1
			}
			change_variable = { name = jen_auto_queue_months subtract = 1 }
			debug_log = "BotD auto-upgrade queue ticking"
		}
		else = {
			jen_auto_upgrade_complete_queued_effect = yes
			remove_variable = jen_auto_queue_building
			remove_variable = jen_auto_queue_months
			debug_log = "BotD auto-upgrade queue finished"
		}
	}
}
